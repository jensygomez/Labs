Script Validacion: |-
  #!/bin/bash
  PUNTOS=0

  echo "=== EVALUANDO TU TRABAJO ==="

  # Validación 1: ¿Está activo? (Independiente de los permisos)
  if systemctl is-active --quiet webcheck.service; then
      echo "✔ [35%] Servicio activo y en ejecución."
      PUNTOS=$((PUNTOS + 35))
  else
      echo "❌ [0%] El servicio no está corriendo."
  fi

  # Validación 2: ¿Está habilitado? (Independiente de si arranca o no)
  if systemctl is-enabled --quiet webcheck.service; then
      echo "✔ [20%] Servicio habilitado al arranque."
      PUNTOS=$((PUNTOS + 20))
  else
      echo "❌ [0%] El servicio está disabled."
  fi

  # Validación 3: ¿El usuario es webapp?
  USER_CHECK=$(systemctl show -p User webcheck.service | cut -d= -f2)
  if [ "$USER_CHECK" = "webapp" ]; then
      echo "✔ [20%] Configurado correctamente para el usuario webapp."
      PUNTOS=$((PUNTOS + 20))
  else
      echo "❌ [0%] El servicio sigue configurado como root u otro usuario."
  fi

  # Validación 4: ¿El directorio de logs es del usuario webapp?
  if [ -d /var/log/webcheck ] && [ "$(stat -c '%U' /var/log/webcheck)" = "webapp" ]; then
      echo "✔ [25%] Permisos de logs corregidos."
      PUNTOS=$((PUNTOS + 25))
  else
      echo "❌ [0%] Error de permisos en el directorio de logs."
  fi

  echo "============================"
  echo "CALIFICACIÓN FINAL: $PUNTOS / 100"
  echo "============================"
